```python
"""Retrospective AF-12 mask selection, trimming and export (uih_mtp.retro_mask)."""
import numpy as np
import pytest

SHAPE = (40, 20)
CENTRE = (20, 10)


def _support():
    """An elliptical support truncated in PE, like the MTP scan's acquired band."""
    pe, spe = np.mgrid[0:SHAPE[0], 0:SHAPE[1]]
    ell = ((pe - CENTRE[0]) / 19.0) ** 2 + ((spe - CENTRE[1]) / 9.5) ** 2 <= 1.0
    return ell & (pe < 30)


def _draw(rng, n):
    """A random pattern with n points anywhere in the plane."""
    m = np.zeros(SHAPE, dtype=bool)
    idx = rng.choice(SHAPE[0] * SHAPE[1], size=n, replace=False)
    m.reshape(-1)[idx] = True
    return m


def test_select_draw_takes_the_smallest_qualifying_draw():
    from uih_mtp.retro_mask import select_draw
    sup = _support()
    rng = np.random.default_rng(0)
    pool = []
    for name, n in (("a", 40), ("b", 400), ("c", 200)):
        while True:
            m = _draw(rng, n)
            if (m & sup).sum() >= 1:
                break
        pool.append((dict(name=name, nnz=int(m.sum())), m))
    counts = {meta["name"]: int((m & sup).sum()) for meta, m in pool}
    target = min(counts["b"], counts["c"])
    meta, mask = select_draw(pool, sup, target)
    assert meta["name"] == min((k for k, v in counts.items() if v >= target), key=lambda k: (counts[k], k))
```

```python
"""Retrospective undersampling masks for the pediatric MTP line (stage-2 spec, section 4).

The house generator (MATLAB `vdPoisMex`) draws on a full rectangle, so a mask that lives inside the
acquired elliptical / partial-Fourier support is built by selection and trimming: take the draw with
the smallest number of points inside the support that still reaches the budget, then drop the points
farthest from k-space centre until exactly the budget remains. Calibration points are protected.
"""
from __future__ import annotations

import glob
import os
from pathlib import Path

import h5py
import numpy as np
import scipy.io

from .bartio import write_cfl


def load_support(npz_path) -> tuple[np.ndarray, np.ndarray, tuple[int, int]]:
    """Acquired support, known zeros and the ACS range from a stage-1 `masks_<c>.npz`."""
    with np.load(str(npz_path)) as d:
        support = np.asarray(d["envelope"], dtype=bool)
        zeros = np.asarray(d["m_zero"], dtype=bool)
        acs = tuple(int(v) for v in np.asarray(d["acs"]).reshape(-1)[:2])
    if (support & zeros).any() or not (support | zeros).all():
        raise ValueError("envelope and m_zero must partition the phase-encode plane")
    return support, zeros, acs


def calib_square(shape, centre, size: int) -> np.ndarray:
    """The `size` x `size` calibration block centred on `centre` (lower half included, as numpy slices)."""
    m = np.zeros(tuple(int(s) for s in shape), dtype=bool)
    h = int(size) // 2
    m[centre[0] - h:centre[0] - h + int(size), centre[1] - h:centre[1] - h + int(size)] = True
    return m


def load_draw_pool(directory) -> list[tuple[dict, np.ndarray]]:
    """Every `draw_*.mat` written by the MATLAB sweep, as (meta, boolean mask)."""
    pool = []
    for p in sorted(glob.glob(os.path.join(str(directory), "draw_*.mat"))):
        d = scipy.io.loadmat(p)
        m = np.asarray(d["mask"]).astype(bool)
        meta = dict(name=os.path.basename(p), nnz=int(m.sum()))
        for k in ("accelx", "accely", "calib", "ellipse", "pp"):
            if k in d:
                meta[k] = round(float(np.asarray(d[k]).squeeze()), 4)
        pool.append((meta, m))
    return pool


def select_draw(pool, support, target: int) -> tuple[dict, np.ndarray]:
    """The draw with the smallest in-support count that still reaches `target` (ties by name)."""
    cands = []
    for meta, m in pool:
        n = int((m & support).sum())
        meta = dict(meta, in_support=n)
        if n >= int(target):
            cands.append((meta, m))
    if not cands:
        best = max((int((m & support).sum()) for _, m in pool), default=0)
        raise ValueError(f"no draw reaches the target {target} inside the support (best {best}); "
                         "extend the accelx sweep downward")
    return min(cands, key=lambda mm: (mm[0]["in_support"], mm[0]["name"]))


def trim_to_budget(mask, support, target: int, centre, protect=None, seed: int = 0) -> np.ndarray:
    """Restrict `mask` to `support` and drop the outermost points until exactly `target` remain.

    Distance is Euclidean from `centre` in index units; ties are broken by a fixed RNG. Points in
    `protect` are never dropped.
    """
    m = np.asarray(mask, dtype=bool) & np.asarray(support, dtype=bool)
    keep = np.zeros_like(m) if protect is None else (np.asarray(protect, dtype=bool) & m)
    n_excess = int(m.sum()) - int(target)
    if n_excess < 0:
        raise ValueError(f"mask has {int(m.sum())} points inside the support, fewer than the target {target}")
    if n_excess == 0:
        return m
    droppable = m & ~keep
    if int(droppable.sum()) < n_excess:
        raise ValueError(f"cannot drop {n_excess} points without touching the protected {int(keep.sum())}")
    pe, spe = np.nonzero(droppable)
    r = np.hypot(pe - centre[0], spe - centre[1])
    rng = np.random.default_rng(seed)
    order = np.lexsort((rng.random(r.size), -r))       # farthest first, random among equal radii
    drop = order[:n_excess]
    out = m.copy()
    out[pe[drop], spe[drop]] = False
    return out


def broadcast_readout(mask2d, n_ro: int) -> np.ndarray:
    """(PE, SPE) -> (RO, PE, SPE) uint8, constant along the fully sampled readout."""
    m = np.asarray(mask2d, dtype=bool)
    return np.broadcast_to(m[None, :, :], (int(n_ro), *m.shape)).astype(np.uint8)


def write_mask_mat(path, mask2d) -> None:
    """MATLAB v5 file, key `mask`, float32 — what the framework's external-mask loader reads."""
    scipy.io.savemat(str(path), {"mask": np.asarray(mask2d, dtype=bool).astype(np.float32)}, format="5")


def write_mask_h5(path, mask3d) -> None:
    with h5py.File(str(path), "w") as f:
        f.create_dataset("mask", data=np.asarray(mask3d, dtype=np.uint8), compression="gzip", compression_opts=1)


def write_support_h5(path, support2d) -> None:
    """Key `support`, uint8 — what `paired_gs.data.known_zero.load_support_map` reads."""
    with h5py.File(str(path), "w") as f:
        f.create_dataset("support", data=np.asarray(support2d, dtype=bool).astype(np.uint8))


def write_pattern_cfl(base: str, mask2d, n_ro: int) -> None:
    """BART sampling pattern for `pics -p`, dims (RO, PE, SPE, 1) in the k-space's own axis order."""
    pat = broadcast_readout(mask2d, n_ro).astype(np.complex64)[..., None]
    Path(base).parent.mkdir(parents=True, exist_ok=True)
    write_cfl(base, pat)
```

```matlab
% prep_mask_draws.m -- pool of house vd-Poisson draws on the (PE, SPE) = (304, 104) plane for the
% pediatric MTP AF-12 study (spec 2026-09-12-16yo-mtp-stage2-af12-design.md, section 4).
% Nothing is chosen here: scripts/prep_mask.py picks the draw whose count inside the acquired
% support first reaches the budget 2635 and trims it to exactly that.
% Isotropic accelx = accely (unlike the 256x176 adult topics' fixed accely = 4): this plane has only
% 104 partitions and a fixed 4x along it would starve the short axis.
% vdPoisMex reseeds from time(0) at ONE-SECOND resolution, hence the pause before every draw.
addpath('/data0/congcong/code/ESPIRiT/utils');
outdir = '/data0/congcong/code/GS/Results_20260908/Peds_TSC_MTP_AF12_retro/data/mask_draws';
if ~exist(outdir, 'dir'), mkdir(outdir); end
calib = 24; ellipse = 0; pp = 1;
accel_list = 2.6:0.05:3.6;
ndraw = 5;
k = 0;
for a = accel_list
    for d = 1:ndraw
        k = k + 1;
        pause(1.1);
        mask = vdPoisMex(304, 104, 304, 104, a, a, calib, ellipse, pp);
        mask = single(mask ~= 0);
        accelx = a; accely = a;
        n = nnz(mask);
```

```python
"""Region-restricted rescoring on the frozen gauge (stage-2 spec, section 7).

The metric MAPS are built exactly as the frozen code builds them — head mask, PSNR peak, SSIM data
range and the LoG kernel all come from the FULL-volume |GT| — and only the averaging region changes.
With `region=None` every number equals the frozen gauge's, which the tests assert.

Frozen sources: diag_artefact/headonly.py (head metrics), paired_gs/metrics/metrics.py (whole-volume
PSNR / SSIM / NMSE), scripts/t1t2_common.py (sharpness). Do not "improve" any definition here.
"""
from __future__ import annotations

import numpy as np
from scipy import ndimage
from skimage.metrics import structural_similarity

from diag_artefact.headonly import HP_SIGMA, head_mask, log_kernel
from diag_artefact.volumes import brain_mask, wm_mask

SSIM_WIN = 7


def _mag(v) -> np.ndarray:
    a = np.asarray(v)
    return (np.abs(a) if np.iscomplexobj(a) else a).astype(np.float32)


def partition_region(shape, axis: int, lo: int, hi: int) -> np.ndarray:
    """Boolean volume selecting `lo <= index < hi` along `axis` (the vendor's central partitions)."""
    r = np.zeros(tuple(int(s) for s in shape), dtype=bool)
    sl = [slice(None)] * len(r.shape)
    sl[int(axis)] = slice(int(lo), int(hi))
    r[tuple(sl)] = True
    return r


def _gradient_magnitude(v: np.ndarray) -> np.ndarray:
    gx = np.zeros_like(v); gy = np.zeros_like(v); gz = np.zeros_like(v)
    gx[:-1] = v[1:] - v[:-1]
    gy[:, :-1] = v[:, 1:] - v[:, :-1]
    gz[:, :, :-1] = v[:, :, 1:] - v[:, :, :-1]
    return np.sqrt(gx * gx + gy * gy + gz * gz + 1e-12)


def score_region(rec, gt, region=None) -> dict:
    """Every metric of the topic tables, averaged over `region` (None = the whole volume)."""
    rec = _mag(rec)
    gt = _mag(gt)
    if rec.shape != gt.shape:
        raise ValueError(f"shape mismatch {rec.shape} vs {gt.shape}")
    reg = np.ones_like(gt, dtype=bool) if region is None else np.asarray(region, dtype=bool)
    head = head_mask(gt) & reg
    gmax = float(gt.max())
    r64, g64 = rec.astype(np.float64), gt.astype(np.float64)

    mse = float(np.mean((r64[head] - g64[head]) ** 2))
    head_psnr = float("inf") if mse == 0 else float(20 * np.log10(gmax) - 10 * np.log10(mse))

    _, smap = structural_similarity(rec, gt, data_range=gmax, full=True)
    head_ssim = float(smap[head].mean())

    k = log_kernel()
    lr = ndimage.convolve(r64 / gmax, k, mode="reflect")
    lg = ndimage.convolve(g64 / gmax, k, mode="reflect")
    den = float(np.linalg.norm(lg[head]))
    head_hfen = float("inf") if den == 0 else float(np.linalg.norm((lr - lg)[head]) / den)

    brain = brain_mask(gt)
    wm = wm_mask(gt, brain) & reg
    if wm.any():
        hp = lambda v: v - ndimage.gaussian_filter(v.astype(np.float32), HP_SIGMA)
        gr = float(np.sqrt(np.mean(hp(rec)[wm] ** 2)))
        gg = float(np.sqrt(np.mean(hp(gt)[wm] ** 2)))
        grain = float("inf") if gg == 0 else gr / gg
    else:
        grain = float("nan")

    # whole-volume gauge (paired_gs.metrics): both magnitudes scaled by max|GT| and clipped to [0, 1]
    scale = max(gmax, 1e-8)
    ru, gu = np.clip(r64 / scale, 0.0, 1.0), np.clip(g64 / scale, 0.0, 1.0)
    mse_w = float(np.mean((ru[reg] - gu[reg]) ** 2))
    psnr = float("inf") if mse_w == 0 else float(-10 * np.log10(mse_w))
    _, smap_w = structural_similarity(ru, gu, data_range=1.0, win_size=SSIM_WIN, channel_axis=None, full=True)
    pad = (SSIM_WIN - 1) // 2
    interior = np.zeros_like(reg)
    interior[pad:-pad, pad:-pad, pad:-pad] = True                  # skimage crops this border from its mean
    ssim = float(smap_w[reg & interior].mean())
    den_n = float(np.sum(g64[reg] ** 2))
    nmse = float("inf") if den_n == 0 else float(np.sum((r64[reg] - g64[reg]) ** 2) / den_n)

    sb = brain & reg
    sharp = float(_gradient_magnitude(rec)[sb].mean() / (_gradient_magnitude(gt)[sb].mean() + 1e-9)) if sb.any() else float("nan")

    return dict(head_psnr=head_psnr, head_ssim=head_ssim, head_hfen=head_hfen, grain_ratio=grain,
                head_frac=float(head.sum() / reg.sum()), gmax=gmax,
                psnr=psnr, ssim=ssim, nmse=nmse, sharp=sharp, n_voxels=int(reg.sum()))
```

Create `tests/test_uih_mtp_bart_pattern.py`:

```python
"""`bart pics -p` is honoured: measured zeros are not the same as unsampled (stage-2 spec, section 5)."""
import shutil
import subprocess

import numpy as np
import pytest

BART = "/home/congcongliu/src/bart/bart"
pytestmark = pytest.mark.skipif(not shutil.which(BART), reason="BART not installed")


def _pics(tmp, ksp, maps, pattern=None, lam=0.001, iters=10):
    from uih_mtp.bartio import read_cfl, write_cfl
    write_cfl(str(tmp / "k"), ksp)
    write_cfl(str(tmp / "m"), maps)
    cmd = [BART, "pics"]
    if pattern is not None:
        write_cfl(str(tmp / "p"), pattern)
        cmd += ["-p", str(tmp / "p")]
    cmd += ["-l1", "-r", str(lam), "-i", str(iters), "-S", str(tmp / "k"), str(tmp / "m"), str(tmp / "out")]
    res = subprocess.run(cmd, capture_output=True, text=True)
    assert res.returncode == 0, res.stdout + res.stderr
    return np.squeeze(read_cfl(str(tmp / "out")))


def _case(seed=0):
    rng = np.random.default_rng(seed)
    nx, ny, nz, nc = 16, 12, 1, 4
    x = rng.standard_normal((nx, ny, nz)) + 1j * rng.standard_normal((nx, ny, nz))
    maps = (rng.standard_normal((nx, ny, nz, nc)) + 1j * rng.standard_normal((nx, ny, nz, nc))).astype(np.complex64)
    maps /= np.sqrt((np.abs(maps) ** 2).sum(-1, keepdims=True))
    k = np.fft.fftshift(np.fft.fftn(np.fft.ifftshift(x[..., None] * maps, axes=(0, 1, 2)),
                                    axes=(0, 1, 2), norm="ortho"), axes=(0, 1, 2))
    omega = np.zeros((nx, ny, nz), dtype=bool)
    omega[:, ::2] = True
    zeros = np.zeros((nx, ny, nz), dtype=bool)
    zeros[:, 1::4] = True                          # "measured zero" columns, disjoint from omega
    return k.astype(np.complex64) * omega[..., None], maps, omega, zeros


def test_pattern_of_the_nonzero_support_reproduces_the_inferred_pattern(tmp_path):
    ksp, maps, omega, _ = _case()
    a = _pics(tmp_path / "a", ksp, maps)
    (tmp_path / "b").mkdir(parents=True, exist_ok=True)
    b = _pics(tmp_path / "b", ksp, maps, pattern=omega.astype(np.complex64)[..., None])
    np.testing.assert_allclose(a, b, rtol=1e-4, atol=1e-5 * np.abs(a).max())


def test_adding_measured_zeros_to_the_pattern_changes_the_result(tmp_path):
    ksp, maps, omega, zeros = _case()
    (tmp_path / "a").mkdir(parents=True, exist_ok=True)
    (tmp_path / "c").mkdir(parents=True, exist_ok=True)
    a = _pics(tmp_path / "a", ksp, maps)
    c = _pics(tmp_path / "c", ksp, maps, pattern=(omega | zeros).astype(np.complex64)[..., None])
    rel = np.linalg.norm(c - a) / np.linalg.norm(a)
    assert rel > 1e-3, rel
```

Create `<TOPIC>/scripts/peds_sched.sh` (the structure of `mcgs_af16_wt/scripts/p3_sched.sh`: only ever an empty card, one job at a time, restartable through the per-job result file):

```bash
#!/usr/bin/env bash
# peds_sched.sh QUEUE -- run the queue one job at a time on an EMPTY allowed card (line rule: card 2 only).
# Queue lines: "classical <c>" | "cssense <c> <lam>" | "jcssense <lam>" | "zsssdu <c>" | "dip <c>" |
#              "inr <c>" | "gs <joint|single_T1w|single_T2s>".  Blank lines and # comments are ignored.
# A line whose result file exists is skipped, so the scheduler can be killed and restarted at any time.
# Touch logs/.sched_pause to hold further launches. Exits when every queue line has its result.
set -uo pipefail
T=/data0/congcong/code/GS/Results_20260908/Peds_TSC_MTP_AF12_retro
W=/data0/congcong/code/GS/mcgs_peds_wt
Q=${1:?queue file}
ALLOWED=${ALLOWED:-"2"}
log() { echo "$(date '+%F %T') $*" | tee -a "$T/logs/sched.log"; }

free_card() {
  for g in $ALLOWED; do
    used=$(nvidia-smi --query-gpu=memory.used --format=csv,noheader,nounits -i "$g" 2>/dev/null | tr -d ' ')
    [[ -z "$used" ]] && continue
    procs=$(nvidia-smi --query-compute-apps=pid --format=csv,noheader -i "$g" 2>/dev/null | grep -c .)
    if [[ "$used" -lt 1500 && "$procs" -eq 0 ]]; then echo "$g"; return 0; fi
  done
  return 1
}

done_file() {                      # done_file <job words...> -> the file that marks it finished
  case "$1" in
    classical) echo "$T/logs/classical_$2.json" ;;
    cssense)   echo "$T/logs/classical_$2_o200_lam$3.json" ;;
    jcssense)  echo "$T/logs/jcssense_o200_lam$2.json" ;;
    zsssdu)    echo "$T/logs/zsssdu_$2.json" ;;
    dip)       echo "$T/logs/dip_$2.json" ;;
    inr)       echo "$T/logs/inr_$2.json" ;;
    gs)        echo "$W/results_arc/peds_af12_$2_score.json" ;;
    *)         echo "" ;;
  esac
}

command_for() {                    # command_for <card> <job words...>
  local g=$1; shift
  case "$1" in
    classical) echo "cd $T && ./scripts/py scripts/recon_classical.py --contrast $2 --gpu $g" ;;
    cssense)   echo "cd $T && ./scripts/py scripts/recon_classical.py --contrast $2 --gpu $g --methods cssense --admm_outer 200 --lam $3 --suffix _o200_lam$3" ;;
    jcssense)  echo "cd $T && ./scripts/py scripts/recon_joint_cssense.py --topic $T --contrasts T1w,T2s --gpu $g --admm_outer 200 --lam $2 --suffix _o200_lam$2" ;;
    zsssdu)    echo "cd $T && ./scripts/py scripts/recon_zsssdu.py --contrast $2 --gpu $g --epochs 100" ;;
    dip)       echo "cd $T && ./scripts/py scripts/recon_dip_inr.py --method dip --contrast $2 --gpu $g --iters 5000 --lr 3e-3 --eval_every 100" ;;
    inr)       echo "cd $T && ./scripts/py scripts/recon_dip_inr.py --method inr --contrast $2 --gpu $g --iters 1500 --eval_every 50" ;;
    gs)        local cfg=configs/peds_af12_T1w_T2s.yaml
               [[ "$2" == single_T1w ]] && cfg=configs/peds_af12_single_T1w.yaml
               [[ "$2" == single_T2s ]] && cfg=configs/peds_af12_single_T2s.yaml
               echo "cd $W && CONFIG=$cfg SAVE_VOLUMES=1 ./autoresearch_peds.sh peds_af12_$2 $g 4000" ;;
  esac
}

log "scheduler up, queue=$Q allowed=[$ALLOWED]"
while true; do
  pending=0; ran=0
  while read -r line || [[ -n "$line" ]]; do
    [[ -z "$line" || "${line:0:1}" == "#" ]] && continue
    set -- $line
    df=$(done_file "$@")
    [[ -n "$df" && -e "$df" ]] && continue
    [[ -z "$df" ]] && { log "bad queue line: $*"; continue; }
    pending=1
    [[ -e "$T/logs/.sched_pause" ]] && { log "paused"; break; }
    card=$(free_card) || break
    cmd=$(command_for "$card" "$@")
    jl="$T/logs/job_$(echo "$*" | tr ' .' '__').log"
    log "launch [$*] on card $card -> $jl"
    bash -c "$cmd" >> "$jl" 2>&1
    rc=$?
    st=MISSING; [[ -e "$df" ]] && st=present
    log "done [$*] rc=$rc result $st"
    printf '%s\t%s\t%s\t%s\n' "$(date '+%F %T')" "$*" "$rc" "$st" >> "$T/logs/sched_status.tsv"
    ran=1
    break                          # re-read the queue from the top after every job
  done < "$Q"
  if [[ "$pending" -eq 0 ]]; then log "ALL DONE"; touch "$T/logs/DONE_SCHED"; break; fi
  [[ "$ran" -eq 0 ]] && sleep 60
done
```

Each job's stdout goes to `<TOPIC>/logs/job_<job>.log`, the scheduler's own progress to `<TOPIC>/logs/sched.log`, and one line per finished job to `<TOPIC>/logs/sched_status.tsv`. The GS jobs run in the worktree, every other job in the topic folder.

Create `<TOPIC>/scripts/rescore.py`:

```python
#!/usr/bin/env python3
"""rescore.py -- the primary table: every arm rescored on the frozen gauge, averaged over the head
inside the vendor's central 80 partitions (SPE 12..92). The secondary table is each arm's own JSON
over all 104 partitions; --full reproduces it here as the cross-check.
"""
import argparse, json, re, sys
from pathlib import Path

import numpy as np

sys.path.insert(0, "/data0/congcong/code/GS/mcgs_peds_wt")
from uih_mtp.rescore import partition_region, score_region

ap = argparse.ArgumentParser()
ap.add_argument("--lo", type=int, default=12)
ap.add_argument("--hi", type=int, default=92)
ap.add_argument("--full", action="store_true", help="score all 104 partitions (the cross-check)")
ap.add_argument("--out", default=None)
args = ap.parse_args()

R = Path(__file__).resolve().parents[1]
CONTRASTS = ("T1w", "T2s")
summary = json.load(open(R / "logs/summary.json"))


def lam_file(c, row_key, prefix):
    lam = next((r.get("lam") for r in summary if r["contrast"] == c and r["method"] == row_key), None)
    if lam is None:
        return None, None
    for p in sorted(R.glob(f"recon/{prefix}_lam*_{c}_complex.npy")):
        m = re.fullmatch(re.escape(prefix) + r"_lam([0-9.]+)_" + c + r"_complex\.npy", p.name)
        if m and abs(float(m.group(1)) - float(lam)) < 1e-12:
            return p, float(lam)
    return None, float(lam)


def arms(c):
    a = [("cgsense", "CG-SENSE", R / f"recon/cgsense_{c}_complex.npy", None)]
    for key, label, prefix in (("cssense_o200", "CS-SENSE", "cssense_o200"),
                               ("jcssense", "Joint SENSE", "jcssense_o200"),
                               ("l1espirit", "L1-ESPIRiT", "l1espirit")):
        p, lam = lam_file(c, key, prefix)
        a.append((key, label, p, lam))
    a += [("zsssdu", "ZS-SSDU", R / f"recon/zsssdu_{c}_complex.npy", None),
          ("dip", "3D-DIP", R / f"recon/dip_{c}_complex.npy", None),
          ("inr", "3D-INR", R / f"recon/inr_{c}_complex.npy", None),
          ("gs_single_ar", "GS-single", R / f"recon/gs_single_ar_{c}_complex.npy", None),
          ("gs_joint_ar", "MC-GS", R / f"recon/gs_joint_ar_{c}_complex.npy", None)]
    return a


out = {}
for c in CONTRASTS:
    gt = np.load(R / f"recon/gt_{c}_complex.npy")
    region = None if args.full else partition_region(gt.shape, 2, args.lo, args.hi)
    for key, label, path, lam in arms(c):
        assert path is not None and Path(path).exists(), f"missing volume for {label} {c}: {path}"
        rec = np.load(path)
        row = score_region(rec, gt, region)
        row.update(method=key, label=label, contrast=c, lam=lam, volume=str(Path(path).name))
        out[f"{key}|{c}"] = row
        print(f"{c} {label:12s} head_psnr {row['head_psnr']:7.3f} ssim {row['head_ssim']:.4f} "
              f"hfen {row['head_hfen']:.4f} psnr {row['psnr']:7.3f} nmse {row['nmse']:.5f} sharp {row['sharp']:.3f}")

p = Path(args.out) if args.out else (R / ("logs/rescore_full.json" if args.full else "logs/rescore.json"))
json.dump(dict(region=("full volume" if args.full else f"head inside partitions {args.lo}..{args.hi}"),
               lo=args.lo, hi=args.hi, rows=out), open(p, "w"), indent=2)
print("wrote", p)
```

```python
#!/usr/bin/env python3
"""readme_tables.py -- the two markdown tables of this topic's README.

Primary  (logs/rescore.json):  every arm on the frozen gauge, averaged over the head inside the
                               vendor's central 80 partitions -- the numbers the README quotes.
Secondary (logs/summary.json): each arm's own full-volume numbers, the cross-check.

Usage: readme_tables.py <topic_dir>
"""
import json
import sys
from pathlib import Path

T = Path(sys.argv[1])
CONTRASTS = ("T1w", "T2s")
WEIGHT = {"T1w": "T1-weighted (FA 16, TE 3.25 ms)", "T2s": "T2*-weighted (FA 4, TE 27.41 ms)"}
LABEL = {"cgsense": "CG-SENSE", "cssense_o200": "CS-SENSE (TV, converged)", "jcssense": "Joint SENSE (JTV)",
         "l1espirit": "L1-ESPIRiT", "zsssdu": "ZS-SSDU", "dip": "3D-DIP", "inr": "3D-INR",
         "gs_single_ar": "GS-single", "gs_joint_ar": "MC-GS"}
ORDER = list(LABEL)
AF = json.load(open(T / "logs/prep_mask.json"))
prim = json.load(open(T / "logs/rescore.json"))
rows_p = prim["rows"]
rows_s = {f"{r['method']}|{r['contrast']}": r for r in json.load(open(T / "logs/summary.json"))}


def fmt(v, p, best=False):
    if v is None:
        return "--"
    t = f"{v:.{p}f}"
    return f"**{t}**" if best else t


def table(c, rows, cols, title):
    sub = [(k, r) for k, r in rows.items() if r.get("contrast") == c and r.get("method") in ORDER]
    if not sub:
        return
    sub.sort(key=lambda kr: -kr[1]["head_psnr"])
    best = {}
    for name, _, hi, _ in cols:
        vals = [r.get(name) for _, r in sub if r.get(name) is not None]
        best[name] = (max(vals) if hi else min(vals)) if vals else None
    print(f"\n**{c}** — {WEIGHT[c]}. {title}\n")
    print("| method | " + " | ".join(h for _, h, _, _ in cols) + " |")
    print("|---" * (len(cols) + 1) + "|")
    for k, r in sub:
        lab = LABEL.get(r["method"], r["method"])
        if r.get("lam"):
            lab += f" (λ {r['lam']:g})"
        cells = [fmt(r.get(n), p, r.get(n) == best[n]) for n, _, _, p in cols]
        print(f"| {lab} | " + " | ".join(cells) + " |")


PRIMARY = [("head_psnr", "head PSNR", True, 3), ("head_ssim", "head SSIM", True, 4),
           ("head_hfen", "head HFEN", False, 4), ("psnr", "PSNR", True, 3),
           ("nmse", "NMSE", False, 5), ("sharp", "sharp", True, 3)]
SECONDARY = [("head_psnr", "head PSNR", True, 3), ("head_ssim", "head SSIM", True, 4),
             ("head_hfen", "head HFEN", False, 4), ("psnr", "PSNR", True, 3), ("nmse", "NMSE", False, 5)]

print(f"### Primary table — {prim['region']}\n")
print(f"Retrospective AF {AF['AF_vs_full_grid']:.2f} of the {AF['plane_points']}-point phase-encode matrix "
      f"({AF['AF_vs_support']:.2f}x of the {AF['n_support']} acquired positions), one mask for both contrasts; "
      f"the partial-Fourier block and the elliptical corners ({AF['n_zero']} positions) are measured zeros for "
      f"every method alike. Best of each column in bold.")
for c in CONTRASTS:
    table(c, rows_p, PRIMARY, "head metrics inside the vendor's central 80 partitions.")

print("\n### Secondary table — full volume, each arm's own scorer\n")
print("The same gauge over all 104 partitions, taken from each arm's own JSON; the cross-check that the "
      "rescorer did not move anything (`scripts/rescore.py --full`).")
for c in CONTRASTS:
    table(c, rows_s, SECONDARY, "full volume.")

print("\n### MC-GS minus GS-single\n")
print("| channel | head PSNR | head SSIM | head HFEN | PSNR |")
print("|---|---|---|---|---|")
for c in CONTRASTS:
    j, s = rows_p.get(f"gs_joint_ar|{c}"), rows_p.get(f"gs_single_ar|{c}")
    if j and s:
        print(f"| {c} | {j['head_psnr']-s['head_psnr']:+.3f} | {j['head_ssim']-s['head_ssim']:+.4f} | "
              f"{j['head_hfen']-s['head_hfen']:+.4f} | {j['psnr']-s['psnr']:+.3f} |")
```
